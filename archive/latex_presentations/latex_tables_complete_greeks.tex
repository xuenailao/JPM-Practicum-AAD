\begin{alertblock}{Critical Observation: Volga Accuracy}
\begin{itemize}
    \item \textbf{Low $\sigma$ (0.15)}: Volga error $\sim130\%$ for all methods
    \item \textbf{Medium $\sigma$ (0.20)}: Volga error $\sim890\%$ (extreme!)
    \item \textbf{High $\sigma$ (0.30)}: Volga error drops to $\sim35\%$ for AAD methods
    \item \textbf{Root cause}: Volga = $\partial^2 V/\partial\sigma^2$ requires finer $\sigma$-grid or Richardson extrapolation
    \item \textbf{Recommendation}: For accurate Volga, use M=101 or implement $\nu$-parametrization
\end{itemize}
\end{alertblock}
