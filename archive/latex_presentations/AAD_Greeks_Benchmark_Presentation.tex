\section{Introduction}

\begin{frame}{Five Greeks Computation Methods}
\begin{itemize}
    \item \textbf{Method 1: Analytical (BSM)} \\
    Closed-form Black-Scholes formulas -- baseline reference

    \item \textbf{Method 2: Bumping} \\
    Finite difference (5 PDE solves) -- simple, widely used

    \item \textbf{Method 3: AAD + Bumping} \\
    AAD for first-order + Bumping for second-order (5 PDE solves)

    \item \textbf{Method 4: Double AAD} \\
    Nested AAD for second-order derivatives (3 PDE solves)

    \item \textbf{Method 5: Edge-Pushing} \\
    Algorithm 4 with Natural Cubic Spline (1 PDE solve) -- \textcolor{blue}{our contribution}
\end{itemize}
\end{frame}

\section{Key Innovations}

\begin{frame}{Technical Contributions}
\begin{enumerate}
    \item \textbf{Natural Cubic Spline Interpolation}
    \begin{itemize}
        \item Replaces linear interpolation ($C^0$) with cubic spline ($C^2$ continuous)
        \item Solves tridiagonal system for second derivatives $M_i$
        \item \textcolor{blue}{Result}: Gamma error reduced from 100\% $\rightarrow$ 2.18\%
    \end{itemize}

    \vspace{0.5em}
    \item \textbf{Fixed Grid Scheme}
    \begin{itemize}
        \item Eliminates adaptive timesteps that depend on $\sigma$
        \item Prevents grid-jumping noise in Volga computation
        \item \textcolor{blue}{Result}: Volga error reduced from 124\% $\rightarrow$ 9\%, 2$\times$ speedup
    \end{itemize}

    \vspace{0.5em}
    \item \textbf{Edge-Pushing Algorithm 4}
    \begin{itemize}
        \item Efficient Hessian computation via adjoint list
        \item Single backward pass for full second-order derivatives
        \item \textcolor{blue}{Result}: 1 PDE solve instead of 5 (Bumping) or 3 (Double AAD)
    \end{itemize}
\end{enumerate}
\end{frame}

% ============================================================================
\begin{frame}{Natural Cubic Spline: Technical Details}
\begin{columns}
\column{0.5\textwidth}
\textbf{Problem:}
\begin{itemize}
    \item Linear interpolation: $V(S_0)$ is piecewise linear
    \item $\Rightarrow \partial^2 V/\partial S_0^2 = 0$ (Gamma = 0!)
\end{itemize}

\vspace{0.5em}
\textbf{Solution:}
\begin{itemize}
    \item Natural cubic spline with $C^2$ continuity
    \item Solve tridiagonal system:
\end{itemize}
\small
$$
\begin{bmatrix}
\frac{h_1 + h_2}{3} & \frac{h_2}{6} & & \\\\
\frac{h_2}{6} & \frac{h_2 + h_3}{3} & \frac{h_3}{6} & \\\\
& \ddots & \ddots & \ddots
\end{bmatrix}
\begin{bmatrix}
M_1 \\\\
M_2 \\\\
\vdots
\end{bmatrix}
=
\begin{bmatrix}
d_1 \\\\
d_2 \\\\
\vdots
\end{bmatrix}
$$

\column{0.5\textwidth}
\textbf{Spline Formula:}
\small
\begin{align*}
V(S_0) &= A \cdot V_i + B \cdot V_{i+1} \\\\
&\quad + \frac{(A^3 - A)h^2}{6} M_i \\\\
&\quad + \frac{(B^3 - B)h^2}{6} M_{i+1}
\end{align*}
where $A = \frac{S_{i+1} - S_0}{h}$, $B = 1 - A$

\vspace{0.5em}
\normalsize
\textbf{Result:}
\begin{itemize}
    \item Gamma error: \textcolor{red}{100\%} $\rightarrow$ \textcolor{blue}{2.18\%}
    \item Maintains global curvature consistency
\end{itemize}
\end{columns}
\end{frame}
